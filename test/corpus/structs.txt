struct Element {
  name: String,
  atomic_number: Scalar,
  density: MassDensity,
}

========================================================================
struct definition with trailing comma
========================================================================

struct Point {
  x: Scalar,
  y: Scalar,
}

---

========================================================================
generic struct definition
========================================================================

struct Vec2<D: Dim> {
  x: D,
  y: D,
}

========================================================================
struct instantiation
========================================================================

let hydrogen = Element {
  name: "Hydrogen",
  atomic_number: 1,
  density: 0.08988 g/L,
}

---

========================================================================
struct instantiation without trailing comma
========================================================================

Point { x: 1, y: 2 }

========================================================================

hydrogen.density
