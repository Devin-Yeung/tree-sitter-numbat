========================================================================
simple function
========================================================================

fn foo(z) = 2 * z + 3

---

========================================================================
function with parameter and return types
========================================================================

fn speed(len: Length, dur: Time) -> Velocity = len / dur

---

========================================================================
function with no parameters
========================================================================

fn f() = 42

---

========================================================================
function declaration without a body
========================================================================

fn print_hello()

---

========================================================================
function with local variables
========================================================================

fn power_4(x: Scalar) = z
  where y = x * x
    and z = y * y

---

========================================================================
function with single local variable
========================================================================

fn square(x) = y
  where y = x * x

---

========================================================================
generic function
========================================================================

fn my_sqrt<T: Dim>(q: T^2) -> T = q^(1/2)

---

========================================================================
generic function without bound
========================================================================

fn identity<T>(x: T) -> T = x

---

========================================================================
generic function with two type parameters
========================================================================

fn first<A, B>(a: A, b: B) -> A = a

---

========================================================================
multiline body
========================================================================

fn body_mass_index(mass: Mass, height: Length) =
    mass / height² -> BMI

---

========================================================================
multiline parameter list
========================================================================

fn add(
  x: Scalar,
  y: Scalar,
) -> Scalar = x + y

========================================================================
function call without arguments
========================================================================

foo()

---

========================================================================
function call with one argument
========================================================================

foo(1)

---

========================================================================
function call with several arguments
========================================================================

mod(17, 4)

---

========================================================================
nested function calls
========================================================================

foo(bar(1), baz(2, 3))

---

========================================================================
multiline function call arguments
========================================================================

foo(
  1,
  2,
  3,
)

---

========================================================================
trailing comma in arguments
========================================================================

foo(1, 2,)

========================================================================
call with expression arguments
========================================================================

assert_eq(oscillation_time(30 cm), 1.1 s, 0.1 s)

passing a function by name
========================================================================

map(inc, xs)

conditional expression
========================================================================

if x >= 0 then 1 else 0

multiline conditional
========================================================================

fn bump(x: Scalar) -> Scalar =
  if x >= 0 && x <= 1
    then 1
    else 0

========================================================================
nested conditional
========================================================================

if a then if b then 1 else 2 else 3

conditional inside parentheses
========================================================================

1 + (if c then 2 else 3)
